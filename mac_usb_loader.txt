/**
 * macOS USB Executable Loader
 * 
 * This program serves as the core of a USB-bootable tool that can
 * launch executables on macOS systems from a portable drive.
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <libgen.h>
#include <sys/stat.h>
#include <pwd.h>
#include <CoreFoundation/CoreFoundation.h>
#include <Security/Authorization.h>

#define MAX_PATH 1024
#define CONFIG_FILE "config/loader.conf"
#define LOG_FILE "logs/loader.log"

typedef struct {
    char executable_dir[MAX_PATH];
    char log_path[MAX_PATH];
    int require_auth;
    char allowed_exts[10][32];
    int num_allowed_exts;
} LoaderConfig;

LoaderConfig config;

/**
 * Get the path to the USB drive where this program is running
 */
char* get_usb_path() {
    static char path[MAX_PATH];
    uint32_t path_size = sizeof(path);
    
    if (_NSGetExecutablePath(path, &path_size) == 0) {
        // Get the directory containing the executable
        char* dir = dirname(path);
        // Assuming the loader is at the root of the USB drive 
        // or in a specific directory structure
        snprintf(path, sizeof(path), "%s", dir);
        return path;
    }
    
    fprintf(stderr, "Failed to determine USB path\n");
    return NULL;
}

/**
 * Load configuration from the config file
 */
int load_config() {
    char config_path[MAX_PATH];
    char usb_path[MAX_PATH];
    
    // Get USB path
    strcpy(usb_path, get_usb_path());
    if (!usb_path) {
        return 0;
    }
    
    // Build config file path
    snprintf(config_path, sizeof(config_path), "%s/%s", usb_path, CONFIG_FILE);
    
    // Set defaults
    strcpy(config.executable_dir, "executables");
    strcpy(config.log_path, "logs/loader.log");
    config.require_auth = 1;
    config.num_allowed_exts = 3;
    strcpy(config.allowed_exts[0], "bin");
    strcpy(config.allowed_exts[1], "sh");
    strcpy(config.allowed_exts[2], "py");
    
    // Read config file
    FILE* fp = fopen(config_path, "r");
    if (!fp) {
        fprintf(stderr, "Warning: Could not open config file at %s, using defaults\n", config_path);
        return 1; // Use defaults
    }
    
    char line[256];
    char key[128], value[128];
    
    while (fgets(line, sizeof(line), fp)) {
        // Skip comments and empty lines
        if (line[0] == '#' || line[0] == '\n') {
            continue;
        }
        
        if (sscanf(line, "%127[^=]=%127[^\n]", key, value) == 2) {
            // Remove whitespace
            char* k = key;
            while (*k == ' ' || *k == '\t') k++;
            char* v = value;
            while (*v == ' ' || *v == '\t') v++;
            
            // Remove trailing whitespace from key
            char* end = k + strlen(k) - 1;
            while (end > k && (*end == ' ' || *end == '\t')) {
                *end = '\0';
                end--;
            }
            
            if (strcmp(k, "executable_dir") == 0) {
                strcpy(config.executable_dir, v);
            } else if (strcmp(k, "log_path") == 0) {
                strcpy(config.log_path, v);
            } else if (strcmp(k, "require_auth") == 0) {
                config.require_auth = (strcmp(v, "true") == 0 || strcmp(v, "1") == 0) ? 1 : 0;
            } else if (strncmp(k, "allowed_ext", 11) == 0) {
                if (config.num_allowed_exts < 10) {
                    strcpy(config.allowed_exts[config.num_allowed_exts], v);
                    config.num_allowed_exts++;
                }
            }
        }
    }
    
    fclose(fp);
    return 1;
}

/**
 * Log message to the log file
 */
void log_message(const char* message) {
    char log_path[MAX_PATH];
    char timestamp[64];
    time_t now = time(NULL);
    strftime(timestamp, sizeof(timestamp), "%Y-%m-%d %H:%M:%S", localtime(&now));
    
    snprintf(log_path, sizeof(log_path), "%s/%s", get_usb_path(), config.log_path);
    
    FILE* fp = fopen(log_path, "a");
    if (fp) {
        fprintf(fp, "[%s] %s\n", timestamp, message);
        fclose(fp);
    }
}

/**
 * Check if file extension is allowed
 */
int is_extension_allowed(const char* filename) {
    const char* ext = strrchr(filename, '.');
    if (!ext) {
        return 0;
    }
    ext++; // Skip the dot
    
    for (int i = 0; i < config.num_allowed_exts; i++) {
        if (strcasecmp(ext, config.allowed_exts[i]) == 0) {
            return 1;
        }
    }
    
    return 0;
}

/**
 * Request user authorization if needed
 */
int request_authorization() {
    if (!config.require_auth) {
        return 1; // No auth required
    }
    
    AuthorizationRef authRef;
    OSStatus status;
    
    status = AuthorizationCreate(NULL, kAuthorizationEmptyEnvironment, 
                                kAuthorizationFlagDefaults, &authRef);
    
    if (status != errAuthorizationSuccess) {
        return 0;
    }
    
    AuthorizationItem item = {kAuthorizationRightExecute, 0, NULL, 0};
    AuthorizationRights rights = {1, &item};
    
    AuthorizationFlags flags = kAuthorizationFlagDefaults | 
                              kAuthorizationFlagInteractionAllowed | 
                              kAuthorizationFlagPreAuthorize | 
                              kAuthorizationFlagExtendRights;
    
    status = AuthorizationCopyRights(authRef, &rights, NULL, flags, NULL);
    AuthorizationFree(authRef, kAuthorizationFlagDefaults);
    
    return (status == errAuthorizationSuccess) ? 1 : 0;
}

/**
 * Execute a program from the USB drive
 */
int execute_program(const char* program_name, int argc, char** argv) {
    char executable_path[MAX_PATH];
    char usb_path[MAX_PATH];
    char log_buffer[MAX_PATH + 256];
    
    strcpy(usb_path, get_usb_path());
    if (!usb_path) {
        return 1;
    }
    
    // Build full path to executable
    snprintf(executable_path, sizeof(executable_path), "%s/%s/%s", 
             usb_path, config.executable_dir, program_name);
    
    // Check if file exists and is executable
    struct stat st;
    if (stat(executable_path, &st) != 0) {
        snprintf(log_buffer, sizeof(log_buffer), "Error: Program %s not found", program_name);
        log_message(log_buffer);
        fprintf(stderr, "%s\n", log_buffer);
        return 1;
    }
    
    if (!(st.st_mode & S_IXUSR)) {
        // Try to make it executable
        if (chmod(executable_path, st.st_mode | S_IXUSR) != 0) {
            snprintf(log_buffer, sizeof(log_buffer), 
                     "Error: Program %s is not executable and couldn't set permissions", program_name);
            log_message(log_buffer);
            fprintf(stderr, "%s\n", log_buffer);
            return 1;
        }
    }
    
    // Check if extension is allowed
    if (!is_extension_allowed(program_name)) {
        snprintf(log_buffer, sizeof(log_buffer), 
                 "Error: Program %s has disallowed extension", program_name);
        log_message(log_buffer);
        fprintf(stderr, "%s\n", log_buffer);
        return 1;
    }
    
    // Request authorization if needed
    if (!request_authorization()) {
        snprintf(log_buffer, sizeof(log_buffer), 
                 "Error: Authorization denied for executing %s", program_name);
        log_message(log_buffer);
        fprintf(stderr, "%s\n", log_buffer);
        return 1;
    }
    
    // Prepare arguments for the new process
    char** new_argv = (char**)malloc((argc + 1) * sizeof(char*));
    if (!new_argv) {
        log_message("Error: Memory allocation failed");
        return 1;
    }
    
    new_argv[0] = executable_path;
    for (int i = 1; i < argc; i++) {
        new_argv[i] = argv[i];
    }
    new_argv[argc] = NULL;
    
    // Log execution
    snprintf(log_buffer, sizeof(log_buffer), "Executing program: %s", program_name);
    log_message(log_buffer);
    
    // Execute the program
    pid_t pid = fork();
    if (pid == 0) {
        // Child process
        execv(executable_path, new_argv);
        // If we get here, execv failed
        exit(1);
    } else if (pid > 0) {
        // Parent process
        int status;
        waitpid(pid, &status, 0);
        
        free(new_argv);
        
        if (WIFEXITED(status)) {
            snprintf(log_buffer, sizeof(log_buffer), 
                     "Program %s exited with status %d", program_name, WEXITSTATUS(status));
            log_message(log_buffer);
            return WEXITSTATUS(status);
        } else {
            snprintf(log_buffer, sizeof(log_buffer), 
                     "Program %s terminated abnormally", program_name);
            log_message(log_buffer);
            return 1;
        }
    } else {
        // Fork failed
        free(new_argv);
        log_message("Error: Fork failed");
        return 1;
    }
}

/**
 * Print usage information
 */
void print_usage(const char* program_name) {
    printf("Usage: %s <executable> [args...]\n", program_name);
    printf("       %s --list\n", program_name);
    printf("       %s --help\n", program_name);
    printf("\nThis tool executes programs from a portable USB drive on macOS systems.\n");
}

/**
 * List available executables
 */
int list_executables() {
    char dir_path[MAX_PATH];
    char usb_path[MAX_PATH];
    
    strcpy(usb_path, get_usb_path());
    if (!usb_path) {
        return 1;
    }
    
    // Build path to executables directory
    snprintf(dir_path, sizeof(dir_path), "%s/%s", usb_path, config.executable_dir);
    
    DIR* dir = opendir(dir_path);
    if (!dir) {
        fprintf(stderr, "Error: Could not open executables directory at %s\n", dir_path);
        return 1;
    }
    
    printf("Available executables:\n");
    printf("----------------------\n");
    
    struct dirent* entry;
    int count = 0;
    
    while ((entry = readdir(dir)) != NULL) {
        // Skip . and ..
        if (strcmp(entry->d_name, ".") == 0 || strcmp(entry->d_name, "..") == 0) {
            continue;
        }
        
        char full_path[MAX_PATH];
        snprintf(full_path, sizeof(full_path), "%s/%s", dir_path, entry->d_name);
        
        struct stat st;
        if (stat(full_path, &st) == 0) {
            // Check if it's a regular file and is executable or has allowed extension
            if (S_ISREG(st.st_mode) && 
                ((st.st_mode & S_IXUSR) || is_extension_allowed(entry->d_name))) {
                printf("  %s\n", entry->d_name);
                count++;
            }
        }
    }
    
    closedir(dir);
    
    if (count == 0) {
        printf("  No executables found\n");
    }
    
    return 0;
}

/**
 * Main entry point
 */
int main(int argc, char** argv) {
    // Load configuration
    if (!load_config()) {
        fprintf(stderr, "Error: Failed to determine USB path or load configuration\n");
        return 1;
    }
    
    // Check command line arguments
    if (argc < 2) {
        print_usage(argv[0]);
        return 1;
    }
    
    // Check for special commands
    if (strcmp(argv[1], "--help") == 0) {
        print_usage(argv[0]);
        return 0;
    } else if (strcmp(argv[1], "--list") == 0) {
        return list_executables();
    }
    
    // Execute the requested program
    return execute_program(argv[1], argc - 1, argv + 1);
}
